// Research-authored UNCOMPILED sketch. Not a complete NTT or native patch.
// Requires the existing canonical gl_reduce/gl_sub/gl_mul helpers.
// Contract: t[i] = s[2*i] + 256*s[2*i+1], i<7; t[7]=s[14], where
// s[k] is the exact byte-diagonal dot of sixteen unsigned 64-bit pairs.
// u32 t words may overlap: do not concatenate or truncate to u128.
// No new masking/randomness, matrix layout, launch or privacy claim.
struct ResearchGoldilocksDot16 {
    static constexpr int modulus_bits = 64;
    static int get_factor() { return 128; }
    // The source table multiplies by 2^get_factor(); reduce removes it.
    __device__ u64 reduce(u32 t0, u32 t1, u32 t2, u32 t3,
                         u32 t4, u32 t5, u32 t6, u32 t7,
                         u64 modulus) const {
        // Precondition modulus == P_GL; owner must bind this publicly.
        const u32 t[8] = {t0,t1,t2,t3,t4,t5,t6,t7};
        u64 lo = 0, hi = 0;
        u32 carry = 0;
        #pragma unroll
        for (int i = 0; i < 8; ++i) {
            u32 s = t[i] + carry; // dot16 bounds imply no u32 overflow
            u64 d = s & 65535u;
            if (i < 4) lo |= d << (16*i);
            else hi |= d << (16*(i-4));
            carry = s >> 16;
        }
        // carry<=15; 2^128 == -2^32 (mod P_GL).
        u64 raw = gl_sub(gl_reduce(lo,hi), u64(carry) << 32);
        // 2^-128 == 2^32-1. The source's 2^128 table factor is removed.
        return gl_mul(raw, EPS_GL);
    }
};
